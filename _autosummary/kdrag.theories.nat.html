kdrag.theories.nat

Algebraic datatype for the Peano natural numbers

Theorem

Statement

nat.add_S

∀x y : Nat, S(x) + y = S(x + y)

nat.add_Z

∀x : Nat, Z + x = x

nat.add_Z_r

∀x : Nat, x + Z = x

nat.add_assoc

∀x y z : Nat, x + (y + z) = x + y + z

nat.add_comm

∀x y : Nat, x + y = y + x

nat.add_s_r

∀x y : Nat, x + S(y) = S(x + y)

nat.add_x_zero

∀x : Nat, x + Z = x

nat.from_to_int

∀n : Nat, from_int(to_int(n)) = n

nat.to_from_int

∀a : ℤ, to_int(from_int(a)) = (if a ≤ 0 then 0 else a)

nat.to_int_S

∀n : Nat, to_int(S(n)) = 1 + to_int(n)

nat.to_int_Z

to_int(Z) = 0

nat.to_int_ge_0

∀n : Nat, to_int(n) ≥ 0
