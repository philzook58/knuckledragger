zf.prod_lemmas.elem_prod_fst_snd_eq

∀A B p : ZFSet,   (p ∈ A × B) = (fst(p) ∈ A ∧   (snd(p) ∈ B ∧ p = pair(fst(p), snd(p))))

zf.prod_lemmas.elem_prod_strong

∀A B p : ZFSet,   (p ∈ A × B) = (∃x y : ZFSet,     x ∈ A ∧ y ∈ B ∧ p = pair(x, y))

zf.prod_lemmas.prod_inter_dist

∀A B X Y : ZFSet, (A ∩ B) × (X ∩ Y) = A × X ∩ B × Y

zf.prod_lemmas.prod_inter_ge

∀A B X Y p : ZFSet,   p ∈ A × X ∩ B × Y → p ∈ (A ∩ B) × (X ∩ Y)

zf.prod_lemmas.prod_inter_le

∀A B X Y p : ZFSet,   p ∈ (A ∩ B) × (X ∩ Y) → p ∈ A × X ∩ B × Y

zf.prod_lemmas.prod_union_left

∀A B X : ZFSet, (A ∪ B) × X = A × X ∪ B × X

zf.prod_lemmas.prod_union_right

∀A B X : ZFSet, X × (A ∪ B) = X × A ∪ X × B

zf.prod_lemmas.rel_le_prod

∀R : ZFSet, is_rel(R) → R ⊆ dom(R) × ran(R)
