algebra.sqmatrix.add_assoc

∀A B C : SqMat, A + (B + C) = A + B + C

algebra.sqmatrix.add_comm

∀A B : SqMat, A + B = B + A

algebra.sqmatrix.add_zero

∀A : SqMat, A + zero = A

algebra.sqmatrix.id_mul

∀A : SqMat, I @ A = A

algebra.sqmatrix.inv_is_inv

∀A : SqMat, is_inv(A) = is_inv(inv(A))

algebra.sqmatrix.inv_left

∀A : SqMat, is_inv(A) → inv(A) @ A = I

algebra.sqmatrix.inv_right

∀A : SqMat, is_inv(A) → A @ inv(A) = I

algebra.sqmatrix.mul_assoc

∀A B C : SqMat, A @ (B @ C) = A @ B @ C

algebra.sqmatrix.mul_id

∀A : SqMat, A @ I = A

algebra.sqmatrix.mul_left_dist

∀A B C : SqMat, A @ (B + C) = A @ B + A @ C

algebra.sqmatrix.mul_right_dist

∀A B C : SqMat, (A + B) @ C = A @ C + B @ C

algebra.sqmatrix.smul_assoc

∀A : SqMat, r s : ℝ, smul(s, smul(r, A)) = smul(s * r, A)

algebra.sqmatrix.smul_comm

∀A : SqMat, s r : ℝ,   smul(r, smul(s, A)) = smul(s, smul(r, A))

algebra.sqmatrix.smul_zero

∀A : SqMat, smul(0, A) = zero

algebra.sqmatrix.trans_idem

∀A : SqMat, trans(trans(A)) = A

algebra.sqmatrix.zero_add

∀A : SqMat, zero + A = A
