kdrag.theories.logic.robinson

Theorem

Statement

logic.robinson.add_succ

∀x y : N, x + succ(y) = succ(x + y)

logic.robinson.add_zero

∀x : N, x + zero = x

logic.robinson.mul_succ

∀x y : N, x * succ(y) = x * y + x

logic.robinson.mul_zero

∀x : N, x * zero = zero

logic.robinson.succ_inj

∀x y : N, succ(x) = succ(y) → x = y

logic.robinson.succ_not_zero

∀x : N, succ(x) ≠ zero

logic.robinson.zero_or_succ

∀y : N, y = zero ∨ (∃x : N, succ(x) = y)
